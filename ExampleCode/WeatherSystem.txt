shader_type canvas_item;

// Master Noise Input Maps
uniform sampler2D cloud_noise1 : repeat_enable;
uniform sampler2D cloud_noise2 : repeat_enable;

// Core Engine Control Channels
uniform float weather_intensity : hint_range(0.0, 1.0) = 0.0;
uniform float lightning_flash : hint_range(0.0, 1.0) = 0.0;
uniform float precipitation_mode : hint_range(0.0, 2.0) = 0.0; // 0=Rain, 1=Snow, 2=Sandstorm
uniform float shelter_factor : hint_range(0.0, 1.0) = 0.0;    // 1.0 = Fully indoors (hides rain/snow)

// Parallax Camera Binding Anchor
uniform vec2 camera_offset = vec2(0.0);

// Canvas Layout Toggles
uniform bool use_pixel_art = false;
uniform vec2 pixel_resolution = vec2(320.0, 180.0);
uniform vec2 wind_vector = vec2(-0.3, 0.7);

uniform_group("Cloud Styling");
uniform vec4 shadow_color : source_color = vec4(0.0, 0.0, 0.1, 0.35);
uniform float shadow_softness : hint_range(0.0, 1.0) = 0.25;

uniform_group("Precipitation Styling");
uniform vec4 rain_color : source_color = vec4(0.6, 0.7, 0.8, 0.25);
uniform vec4 snow_color : source_color = vec4(0.95, 0.98, 1.0, 0.7);
uniform vec4 sandstorm_color : source_color = vec4(0.76, 0.62, 0.45, 0.5);
uniform float density = 50.0;

vec2 hash22(vec2 p) {
    p = vec2(dot(p, vec2(127.1, 311.7)), dot(p, vec2(269.5, 183.3)));
    return fract(sin(p) * 43758.5453123);
}

void fragment() {
    // 1. Retro Pixel-Art Grid Rasterization Snap
    vec2 base_uv = UV;
    if (use_pixel_art) {
        base_uv = floor(UV * pixel_resolution) / pixel_resolution;
    }

    // Bind camera offsets dynamically to achieve high-performance world parallax depth layers
    vec2 parallax_uv = base_uv + (camera_offset * 0.0005);

    // 2. Cloud Shadows Tracking Loop
    vec2 cloud_uv1 = parallax_uv + (wind_vector * 0.03) * TIME;
    vec2 cloud_uv2 = parallax_uv + (wind_vector * 0.015) * TIME + vec2(0.3, 0.2);
    
    float n1 = texture(cloud_noise1, cloud_uv1).r;
    float n2 = texture(cloud_noise2, cloud_uv2).r;
    float cloud_mask = smoothstep(mix(0.7, 0.3, weather_intensity), mix(0.7, 0.3, weather_intensity) + shadow_softness, n1 * n2);
    
    vec4 base_shadow = vec4(shadow_color.rgb, cloud_mask * shadow_color.a * clamp(weather_intensity * 2.0, 0.0, 1.0));
    vec4 lightning_glow = vec4(0.9, 0.95, 1.0, 0.6) * (1.0 - cloud_mask);
    vec4 final_shadow = mix(base_shadow, lightning_glow, lightning_flash);

    // Master operational activation boundaries
    float global_activation = smoothstep(0.3, 0.7, weather_intensity) * (1.0 - shelter_factor);

    // 3. Game-Ready 2D Grid Cellular Rain System (Fixed Angle Rotation Matrix)
    float rain_angle = atan(wind_vector.x, wind_vector.y);
    mat2 rain_rot = mat2(vec2(cos(rain_angle), -sin(rain_angle)), vec2(sin(rain_angle), cos(rain_angle)));
    vec2 rotated_uv = rain_rot * (parallax_uv - 0.5) + 0.5;
    
    vec2 grid_uv = vec2(rotated_uv.x * density, rotated_uv.y * (density * 0.15));
    vec2 cell_id = floor(grid_uv);
    vec2 cell_rand = hash22(cell_id);
    
    float falling_speed = 4.0 + cell_rand.y * 3.0;
    float vertical_cycle = fract(grid_uv.y - TIME * falling_speed + cell_rand.x);
    
    float drop_x = smoothstep(0.08, 0.0, abs(fract(grid_uv.x) - 0.5 - (cell_rand.x - 0.5) * 0.4));
    float drop_y = smoothstep(0.7, 0.0, vertical_cycle);
    float rain_needle_mask = drop_x * drop_y * step(0.9, cell_rand.y);
    
    vec4 dynamic_rain_color = mix(rain_color, vec4(1.0, 1.0, 1.0, 0.6), lightning_flash);
    vec4 final_rain = vec4(dynamic_rain_color.rgb, rain_needle_mask * rain_color.a * 1.5 * global_activation);

    // 4. Procedural Floor Splash Impacts
    vec2 splash_uv = parallax_uv * (density * 0.4);
    vec2 splash_id = floor(splash_uv);
    vec2 splash_frac = fract(splash_uv) - 0.5;
    float splash_time = fract(TIME * 2.5 + hash22(splash_id).x);
    float splash_radius = splash_time * 0.4;
    float dist = length(splash_frac);
    float ring = smoothstep(splash_radius - 0.04, splash_radius, dist) - smoothstep(splash_radius, splash_radius + 0.04, dist);
    float splash_mask = step(0.94 - (weather_intensity * 0.04), hash22(splash_id + vec2(11.0)).y) * ring * (1.0 - splash_time);
    vec4 final_splashes = vec4(dynamic_rain_color.rgb, splash_mask * rain_color.a * 0.6 * global_activation);
    vec4 full_rain_set = mix(final_rain, final_splashes, final_splashes.a);

    // 5. Swaying Snow Flakes Engine Loop
    vec2 snow_drift = vec2(wind_vector.x * 0.5 + sin(TIME + parallax_uv.y * 4.0) * 0.15, wind_vector.y * 0.3);
    vec2 snow_uv = parallax_uv + snow_drift * TIME * 0.8;
    vec2 scaled_snow_uv = snow_uv * (density * 0.6);
    vec2 snow_id = floor(scaled_snow_uv);
    vec2 snow_frac = fract(scaled_snow_uv) - 0.5;
    float snow_hash = hash22(snow_id).x;
    float flake_shape = smoothstep(0.15 + (snow_hash * 0.1), 0.0, length(snow_frac));
    float flake_mask = step(0.92 - (weather_intensity * 0.06), snow_hash) * flake_shape;
    vec4 final_snow = vec4(snow_color.rgb, flake_mask * snow_color.a * global_activation);

    // 6. Volumetric Atmospheric Sandstorm Overlay
    vec2 sand_uv = parallax_uv + vec2(wind_vector.x * 1.5, wind_vector.y * 0.1) * TIME;
    float sand_noise = texture(cloud_noise1, sand_uv * 0.5).r * texture(cloud_noise2, sand_uv * 0.25).g;
    vec4 final_sand = vec4(sandstorm_color.rgb, sand_noise * sandstorm_color.a * weather_intensity * (1.0 - shelter_factor * 0.7));

    // Channel Switching Router Matrix
    vec4 precip_layer = vec4(0.0);
    if (precipitation_mode < 0.5) {
        precip_layer = full_rain_set;
    } else if (precipitation_mode < 1.5) {
        precip_layer = final_snow;
    } else {
        precip_layer = final_sand;
    }
    
    COLOR = mix(final_shadow, precip_layer, precip_layer.a);
}
extends ColorRect
class_name WeatherSystem

enum WeatherState { SUNNY, OVERCAST, RAIN_STORM, SNOW_BLIZZARD, SANDSTORM }

@export_group("System Base Engine")
@export var current_state: WeatherState = WeatherState.SUNNY
@export var transition_duration: float = 6.0

@export_group("Retro Settings")
@export var pixel_art_mode: bool = false
@export var native_resolution: Vector2 = Vector2(320, 180)

@export_group("Wind Vectors")
@export var base_wind: Vector2 = Vector2(-0.2, 0.7)
@export var gust_interval: float = 4.0

@export_group("Lightning Generator Rig")
@export var min_strike_interval: float = 8.0
@export var max_strike_interval: float = 22.0

@export_group("Hardware Hook Link")
@export var camera_node: Camera2D

@export_group("Audio Engine Configurations")
@export var wind_loop_sound: AudioStream
@export var rain_loop_sound: AudioStream
@export var sandstorm_loop_sound: AudioStream
@export var thunder_crack_sound: AudioStream
@export var thunder_rumble_sound: AudioStream

# API Engine Metrics Exposed to Survival Scripts
var current_temperature: float = 25.0  
var wetness_factor: float = 0.0        
var visibility_range: float = 1.0      
var inside_shelter: bool = false # Toggle TRUE from your maps to stop indoor rain instantly

var _strike_timer: Timer
var _gust_timer: Timer
var _shader_mat: ShaderMaterial

var _wind_player: AudioStreamPlayer
var _rain_player: AudioStreamPlayer
var _sand_player: AudioStreamPlayer
var _thunder_player: AudioStreamPlayer

var _camera_shake_intensity: float = 0.0
var _camera_shake_decay: float = 10.0
var _target_wind: Vector2 = Vector2.ZERO

func _ready() -> void {
	anchors_preset = PRESET_FULL_RECT
	mouse_filter = MOUSE_FILTER_IGNORE
	
	_shader_mat = ShaderMaterial.new()
	_shader_mat.shader = load("res://ultimate_weather_engine.gdshader")
	material = _shader_mat
	
	_shader_mat.set_shader_parameter("cloud_noise1", _generate_procedural_noise(12345))
	_shader_mat.set_shader_parameter("cloud_noise2", _generate_procedural_noise(67890))
	_target_wind = base_wind
	
	_strike_timer = Timer.new(); _strike_timer.one_shot = true
	_strike_timer.timeout.connect(_on_strike_timer_timeout); add_child(_strike_timer)
	
	_gust_timer = Timer.new(); _gust_timer.timeout.connect(_on_gust_timer_timeout)
	add_child(_gust_timer); _gust_timer.start(gust_interval)
	
	_initialize_audio_nodes()
	
	_shader_mat.set_shader_parameter("use_pixel_art", pixel_art_mode)
	_shader_mat.set_shader_parameter("pixel_resolution", native_resolution)
	
	change_weather(current_state, true)

func _process(delta: float) -> void {
	# 1. Parallax Matrix Bind: Sync world camera position directly with the shader
	if camera_node:
		_shader_mat.set_shader_parameter("camera_offset", camera_node.global_position)
		
	# 2. Smooth Shelter Transition (Slides between 0.0=outdoors and 1.0=indoors)
	var target_shelter: float = 1.0 if inside_shelter else 0.0
	var current_shelter: float = _shader_mat.get_shader_parameter("shelter_factor")
	_shader_mat.set_shader_parameter("shelter_factor", move_toward(current_shelter, target_shelter, delta * 3.5))

	# 3. Wind Vector Swell Interpolation Loops
	var current_wind: Vector2 = _shader_mat.get_shader_parameter("wind_vector")
	_shader_mat.set_shader_parameter("wind_vector", current_wind.lerp(_target_wind, delta * 1.5))
	
	# 4. Mechanical Screen Shake Processing Engine Thread
	if camera_node and _camera_shake_intensity > 0.0:
		_camera_shake_intensity = move_toward(_camera_shake_intensity, 0.0, _camera_shake_decay * delta)
		camera_node.offset = Vector2(randf_range(-_camera_shake_intensity, _camera_shake_intensity), randf_range(-_camera_shake_intensity, _camera_shake_intensity))
	elif camera_node:
		camera_node.offset = Vector2.ZERO
		
	_update_audio_volumes()
	_calculate_survival_metrics(delta)

func change_weather(new_state: WeatherState, instant: bool = false) -> void {
	current_state = new_state
	var target_intensity: float = 0.0
	var target_mode: float = 0.0 
	
	match current_state:
		WeatherState.SUNNY:
			target_intensity = 0.0
			_strike_timer.stop()
		WeatherState.OVERCAST:
			target_intensity = 0.45
			_strike_timer.stop()
		WeatherState.RAIN_STORM:
			target_intensity = 1.0
			target_mode = 0.0
			_schedule_next_strike()
		WeatherState.SNOW_BLIZZARD:
			target_intensity = 1.0
			target_mode = 1.0
			_strike_timer.stop()
		WeatherState.SANDSTORM:
			target_intensity = 1.0
			target_mode = 2.0
			_strike_timer.stop()

	var duration = 0.01 if instant else transition_duration
	var tween = create_tween().set_parallel(true)
	tween.tween_property(_shader_mat, "shader_parameter/weather_intensity", target_intensity, duration).set_trans(Tween.TRANS_SINE)
	tween.tween_property(_shader_mat, "shader_parameter/precipitation_mode", target_mode, duration).set_trans(Tween.TRANS_SINE)

func _initialize_audio_nodes() -> void {
	_wind_player = AudioStreamPlayer.new(); _wind_player.stream = wind_loop_sound; _wind_player.volume_db = -80.0; add_child(_wind_player)
	_rain_player = AudioStreamPlayer.new(); _rain_player.stream = rain_loop_sound; _rain_player.volume_db = -80.0; add_child(_rain_player)
	_sand_player = AudioStreamPlayer.new(); _sand_player.stream = sandstorm_loop_sound; _sand_player.volume_db = -80.0; add_child(_sand_player)
	_thunder_player = AudioStreamPlayer.new(); add_child(_thunder_player)
	
	if wind_loop_sound: _wind_player.play()
	if rain_loop_sound: _rain_player.play()
	if sandstorm_loop_sound: _sand_player.play()

func _update_audio_volumes() -> void {
	var intensity: float = _shader_mat.get_shader_parameter("weather_intensity")
	var mode: float = _shader_mat.get_shader_parameter("precipitation_mode")
	var shelter: float = _shader_mat.get_shader_parameter("shelter_factor")
	
	# Muffle loops slightly when the user steps inside a covered structure
	var audio_shelter_muffle = lerp(1.0, 0.25, shelter)
	
	_wind_player.volume_db = linear_to_db(clamp(mix(0.01, 0.85, intensity) * audio_shelter_muffle, 0.0001, 1.0)) if intensity > 0.1 else -80.0
	_rain_player.volume_db = linear_to_db(clamp(mix(0.01, 0.9, intensity) * audio_shelter_muffle, 0.0001, 1.0)) if (intensity > 0.4 and mode < 0.5) else -80.0
	_sand_player.volume_db = linear_to_db(clamp(mix(0.01, 0.95, intensity) * audio_shelter_muffle, 0.0001, 1.0)) if (intensity > 0.4 and mode > 1.5) else -80.0

func _calculate_survival_metrics(delta: float) -> void {
	match current_state:
		WeatherState.SUNNY:
			current_temperature = move_toward(current_temperature, 25.0, delta * 2.0)
			wetness_factor = move_toward(wetness_factor, 0.0, delta * 0.1)
			visibility_range = move_toward(visibility_range, 1.0, delta * 0.5)
		WeatherState.OVERCAST:
			current_temperature = move_toward(current_temperature, 19.0, delta * 1.0)
			wetness_factor = move_toward(wetness_factor, 0.0, delta * 0.1)
			visibility_range = move_toward(visibility_range, 0.8, delta * 0.3)
		WeatherState.RAIN_STORM:
			current_temperature = move_toward(current_temperature, 14.0, delta * 1.0)
			wetness_factor = move_toward(wetness_factor, 1.0 if not inside_shelter else wetness_factor, delta * 0.2)
			visibility_range = move_toward(visibility_range, 0.6, delta * 0.2)
		WeatherState.SNOW_BLIZZARD:
			current_temperature = move_toward(current_temperature, -15.0 if not inside_shelter else 10.0, delta * 3.0)
			wetness_factor = move_toward(wetness_factor, 0.0, delta * 0.2)
			visibility_range = move_toward(visibility_range, 0.2, delta * 0.4)
		WeatherState.SANDSTORM:
			current_temperature = move_toward(current_temperature, 42.0 if not inside_shelter else 24.0, delta * 4.0)
			wetness_factor = move_toward(wetness_factor, 0.0, delta * 0.5)
			visibility_range = move_toward(visibility_range, 0.15, delta * 0.5)

func _on_gust_timer_timeout() -> void {
	if current_state == WeatherState.RAIN_STORM:
		_target_wind = Vector2(randf_range(-0.6, 0.1), randf_range(0.7, 1.1))
	elif current_state == WeatherState.SNOW_BLIZZARD:
		_target_wind = Vector2(randf_range(-1.2, -0.6), randf_range(0.3, 0.6))
	elif current_state == WeatherState.SANDSTORM:
		_target_wind = Vector2(randf_range(-1.8, -1.0), randf_range(-0.2, 0.2))
	else:
		_target_wind = base_wind

func _trigger_lightning_flash() -> void {
	var bolt_strength: float = randf_range(0.4, 1.0)
	var tween = create_tween()
	tween.tween_property(_shader_mat, "shader_parameter/lightning_flash", bolt_strength, 0.04).set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	tween.tween_property(_shader_mat, "shader_parameter/lightning_flash", bolt_strength * 0.15, 0.05)
	tween.tween_property(_shader_mat, "shader_parameter/lightning_flash", bolt_strength * 0.75, 0.03)
	tween.tween_property(_shader_mat, "shader_parameter/lightning_flash", 0.0, 0.5 * bolt_strength).set_trans(Tween.TRANS_CUBIC).set_ease(Tween.EASE_IN)
		
	if camera_node:
		_camera_shake_intensity = lerp(3.0, 16.0, bolt_strength)
		_camera_shake_decay = lerp(22.0, 9.0, bolt_strength)
	
	get_tree().create_timer(lerp(2.2, 0.15, bolt_strength)).timeout.connect(func(): _play_procedural_thunder(bolt_strength))

func _play_procedural_thunder(strength: float) -> void {
	if current_state != WeatherState.RAIN_STORM: return
	_thunder_player.stream = thunder_crack_sound if strength > 0.78 else thunder_rumble_sound
	_thunder_player.volume_db = linear_to_db(clamp(mix(0.3, 1.0, strength), 0.0001, 1.0))
	_thunder_player.play()

func _generate_procedural_noise(target_seed: int) -> NoiseTexture2D:
	var noise_tex = NoiseTexture2D.new(); noise_tex.seamless = true; noise_tex.repeat = true
	var lite_noise = FastNoiseLite.new(); lite_noise.seed = target_seed; lite_noise.noise_type = FastNoiseLite.TYPE_SIMPLEX; lite_noise.frequency = 0.008
	noise_tex.noise = lite_noise
	return noise_tex

func _schedule_next_strike() -> void { if current_state == WeatherState.RAIN_STORM: _strike_timer.start(randf_range(min_strike_interval, max_strike_interval)) }
func _on_strike_timer_timeout() -> void { if current_state == WeatherState.RAIN_STORM: _trigger_lightning_flash(); _schedule_next_strike() }

This forces the shader's internal rotation matrix to align the rain needles exactly with your isometric camera perspective, preventing the rain from looking out-of-place against angled geometry.🏃 3.
